def cek_tahun(tahun):
    # tahun jadi angka tahun nol atau negatif eror tidak valid
    try:
        tahun = int(tahun)
    except ValueError:
        raise DataFilmError("Tahun harus berupa angka")
    if tahun <= 0:
        raise DataFilmError("Tahun harus lebih dari 0")
    return tahun
